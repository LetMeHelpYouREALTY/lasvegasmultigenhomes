## 🎨 PAGE TEMPLATE STRUCTURE

Every new page should follow this structure:

```html
<!doctype html>
<html lang="en">
<head>
  <!-- SEO Meta Tags -->
  <meta charset="utf-8">
  <title>Primary Keyword | Las Vegas Multi-Gen Homes</title>
  <meta name="description" content="150-160 character description with keyword">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  
  <!-- Canonical & Robots -->
  <link rel="canonical" href="https://www.lasvegasmultigenhomes.com/page.html">
  <meta name="robots" content="index, follow">
  
  <!-- Open Graph -->
  <meta property="og:title" content="Page Title">
  <meta property="og:description" content="Page description">
  <meta property="og:image" content="image-url.jpg">
  <meta property="og:url" content="https://www.lasvegasmultigenhomes.com/page.html">
  
  <!-- Schema.org JSON-LD -->
  <script type="application/ld+json">
  {
    "@context": "https://schema.org",
    "@type": "WebPage",
    "name": "Page Title",
    "description": "Page description",
    "url": "https://www.lasvegasmultigenhomes.com/page.html"
  }
  </script>
  
  <!-- CSS (reuse existing) -->
  <link rel="stylesheet" href="plugins/bootstrap-4.6.2/css/bootstrap.min.css">
  <link rel="stylesheet" href="views/templates/_common/css/common.css">
</head>

<body>
  <!-- Navigation (consistent across site) -->
  
  <!-- Hero Section -->
  <section class="hero">
    <h1>Primary Keyword H1</h1>
  </section>
  
  <!-- Main Content -->
  <section class="content">
    <article>
      <!-- Introduction with keyword -->
      <!-- Content sections with H2/H3 -->
      <!-- Images with alt text -->
      <!-- Internal links -->
    </article>
  </section>
  
  <!-- Call-to-Action -->
  <section class="cta">
    <h2>Ready to Find Your Multi-Gen Home?</h2>
    <a href="contact.html" class="btn">Contact Dr. Jan Duffy</a>
    <a href="tel:702-550-0112" class="btn">Call 702-550-0112</a>
  </section>
  
  <!-- Related Content -->
  <section class="related">
    <h2>Related Resources</h2>
    <!-- Links to 3-4 related pages -->
  </section>
  
  <!-- FAQ Section (for featured snippets) -->
  <section class="faq">
    <h2>Frequently Asked Questions</h2>
    <!-- 5-8 Q&A pairs with Schema markup -->
  </section>
  
  <!-- Footer (consistent across site) -->
  
  <!-- Scripts -->
</body>
</html>
```
